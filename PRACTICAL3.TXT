using System;
using System.Collections.Generic;
using System.Linq;

class Expense
{
    public int Id { get; set; }
    public double Amount { get; set; }
    public string Category { get; set; }
    public string Description { get; set; }
    public DateTime Date { get; set; }

    public Expense(int id, double amount, string category,
                   string description, DateTime date)
    {
        Id = id;
        Amount = amount;
        Category = category;
        Description = description;
        Date = date;
    }

    public void Display()
    {
        Console.WriteLine(
            $"ID: {Id} | Amount: ₹{Amount:F2} | " +
            $"Category: {Category} | Description: {Description} | " +
            $"Date: {Date:dd-MM-yyyy}"
        );
    }
}

class ExpenseTracker
{
    private List<Expense> expenses = new List<Expense>();
    private int nextId = 1;

    // Add Expense
    public void AddExpense()
    {
        try
        {
            Console.Write("Enter Amount: ");
            double amount = Convert.ToDouble(Console.ReadLine());

            if (amount <= 0)
                throw new ArgumentException(
                    "Amount must be greater than zero.");

            Console.Write("Enter Category: ");
            string category = Console.ReadLine();

            if (string.IsNullOrWhiteSpace(category))
                throw new ArgumentException(
                    "Category cannot be empty.");

            Console.Write("Enter Description: ");
            string description = Console.ReadLine();

            if (string.IsNullOrWhiteSpace(description))
                description = "No description";

            Expense expense = new Expense(
                nextId++,
                amount,
                category,
                description,
                DateTime.Now
            );

            expenses.Add(expense);

            Console.WriteLine("\nExpense added successfully!");
        }
        catch (FormatException)
        {
            Console.WriteLine(
                "Error: Please enter a valid numeric amount.");
        }
        catch (ArgumentException ex)
        {
            Console.WriteLine("Error: " + ex.Message);
        }
        catch (Exception ex)
        {
            Console.WriteLine("Unexpected Error: " + ex.Message);
        }
    }

    // View All Expenses
    public void ViewExpenses()
    {
        try
        {
            if (expenses.Count == 0)
                throw new Exception("No expenses available.");

            Console.WriteLine("\n========== ALL EXPENSES ==========");

            foreach (Expense expense in expenses)
            {
                expense.Display();
            }

            Console.WriteLine("==================================");
        }
        catch (Exception ex)
        {
            Console.WriteLine("Error: " + ex.Message);
        }
    }

    // Total Expense
    public void ShowTotal()
    {
        try
        {
            if (expenses.Count == 0)
                throw new Exception("No expenses available.");

            double total = expenses.Sum(e => e.Amount);

            Console.WriteLine(
                $"\nTotal Expense = ₹{total:F2}");
        }
        catch (Exception ex)
        {
            Console.WriteLine("Error: " + ex.Message);
        }
    }

    // Search by Category
    public void SearchByCategory()
    {
        try
        {
            Console.Write("Enter Category: ");
            string category = Console.ReadLine();

            var result = expenses.Where(
                e => e.Category.Equals(
                    category,
                    StringComparison.OrdinalIgnoreCase
                )
            ).ToList();

            if (result.Count == 0)
                throw new Exception(
                    "No expenses found for this category.");

            Console.WriteLine(
                $"\nExpenses in category: {category}");

            foreach (Expense expense in result)
            {
                expense.Display();
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine("Error: " + ex.Message);
        }
    }

    // Delete Expense
    public void DeleteExpense()
    {
        try
        {
            Console.Write("Enter Expense ID to delete: ");
            int id = Convert.ToInt32(Console.ReadLine());

            Expense expense = expenses.FirstOrDefault(
                e => e.Id == id);

            if (expense == null)
                throw new Exception("Expense ID not found.");

            expenses.Remove(expense);

            Console.WriteLine(
                "Expense deleted successfully!");
        }
        catch (FormatException)
        {
            Console.WriteLine("Error: Enter a valid ID.");
        }
        catch (Exception ex)
        {
            Console.WriteLine("Error: " + ex.Message);
        }
    }
}

class Program
{
    static void Main()
    {
        ExpenseTracker tracker = new ExpenseTracker();

        while (true)
        {
            Console.WriteLine("\n================================");
            Console.WriteLine("       EXPENSE TRACKER");
            Console.WriteLine("================================");
            Console.WriteLine("1. Add Expense");
            Console.WriteLine("2. View All Expenses");
            Console.WriteLine("3. Show Total Expense");
            Console.WriteLine("4. Search by Category");
            Console.WriteLine("5. Delete Expense");
            Console.WriteLine("6. Exit");
            Console.WriteLine("================================");

            try
            {
                Console.Write("Enter your choice: ");
                int choice = Convert.ToInt32(Console.ReadLine());

                switch (choice)
                {
                    case 1:
                        tracker.AddExpense();
                        break;

                    case 2:
                        tracker.ViewExpenses();
                        break;

                    case 3:
                        tracker.ShowTotal();
                        break;

                    case 4:
                        tracker.SearchByCategory();
                        break;

                    case 5:
                        tracker.DeleteExpense();
                        break;

                    case 6:
                        Console.WriteLine(
                            "Thank you for using Expense Tracker!");
                        return;

                    default:
                        Console.WriteLine(
                            "Invalid choice! Select 1 to 6.");
                        break;
                }
            }
            catch (FormatException)
            {
                Console.WriteLine(
                    "Error: Please enter a valid number.");
            }
        }
    }
}
